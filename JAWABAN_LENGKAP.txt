1. FETCH ALL CHATS & SYNC KE DATABASE

```php
<?php

namespace App\Services;

use Illuminate\Support\Facades\Http;
use App\Models\Donatur;
use App\Models\Message;

class WhatsAppSyncService
{
    private $apiUrl = 'http://72.62.125.132/api/v1';
    private $apiKey = 'wapi_35a798b44e8fed9075487b003bb8df8197ffb69d3805f17a379d26495de43af3';
    private $sessionId = '05786992-9a96-4bf1-b71f-f2f4f28488c6';

    public function syncAllChats()
    {
        $offset = 0;
        $limit = 50;
        $hasMore = true;

        while ($hasMore) {
            $response = Http::withHeaders([
                'X-API-Key' => $this->apiKey
            ])->get("{$this->apiUrl}/sessions/{$this->sessionId}/chats", [
                'limit' => $limit,
                'offset' => $offset,
                'filter' => 'personal' // Only personal chats, exclude groups
            ]);

            if ($response->successful()) {
                $data = $response->json()['data'];
                
                foreach ($data['chats'] as $chat) {
                    // Save or update donatur
                    Donatur::updateOrCreate(
                        ['whatsapp_id' => $chat['id']],
                        [
                            'name' => $chat['name'],
                            'phone' => $chat['phone'],
                            'last_message_at' => \Carbon\Carbon::createFromTimestampMs($chat['timestamp']),
                            'message_count' => $chat['messageCount'],
                            'is_group' => $chat['isGroup']
                        ]
                    );
                }

                $hasMore = $data['hasMore'];
                $offset += $limit;
            } else {
                break;
            }
        }

        return "Synced chats successfully";
    }

    public function syncChatMessages($chatId)
    {
        $offset = 0;
        $limit = 100;
        $hasMore = true;

        while ($hasMore) {
            $response = Http::withHeaders([
                'X-API-Key' => $this->apiKey
            ])->get("{$this->apiUrl}/messages/{$this->sessionId}/chats/{$chatId}/messages", [
                'limit' => $limit,
                'offset' => $offset
            ]);

            if ($response->successful()) {
                $data = $response->json()['data'];
                
                foreach ($data['messages'] as $message) {
                    Message::updateOrCreate(
                        ['whatsapp_message_id' => $message['id']],
                        [
                            'donatur_id' => Donatur::where('whatsapp_id', $chatId)->first()->id,
                            'content' => $message['body'],
                            'from_me' => $message['fromMe'],
                            'type' => $message['type'],
                            'status' => $message['status'],
                            'timestamp' => \Carbon\Carbon::createFromTimestampMs($message['timestamp']),
                            'has_media' => $message['hasMedia'],
                            'media_url' => $message['mediaUrl'],
                            'media_type' => $message['mediaType']
                        ]
                    );
                }

                $hasMore = $data['hasMore'];
                $offset += $limit;
            } else {
                break;
            }
        }

        return "Synced messages for chat {$chatId}";
    }
}
```

        // Validate session ID (security check)
        if ($sessionId !== '05786992-9a96-4bf1-b71f-f2f4f28488c6') {
            return response()->json(['error' => 'Invalid session'], 403);
        }

        switch ($event) {
            case 'messages.upsert':
                $this->handleNewMessage($data);
                break;
            
            case 'messages.update':
                $this->handleMessageUpdate($data);
                break;
            
            case 'connection.update':
                $this->handleConnectionUpdate($data);
                break;
        }

        return response()->json(['success' => true]);
    }

    private function handleNewMessage($data)
    {
        foreach ($data['messages'] as $message) {
            $chatId = $message['key']['remoteJid'];
            $messageId = $message['key']['id'];
            $fromMe = $message['key']['fromMe'];

            // Get or create donatur
            $donatur = Donatur::firstOrCreate(
                ['whatsapp_id' => $chatId],
                [
                    'name' => $message['pushName'] ?? 'Unknown',
                    'phone' => str_replace(['@s.whatsapp.net', '@c.us'], '', $chatId)
                ]
            );

            // Save message
            Message::updateOrCreate(
                ['whatsapp_message_id' => $messageId],
                [
                    'donatur_id' => $donatur->id,
                    'content' => $message['message']['conversation'] ?? '',
                    'from_me' => $fromMe,
                    'type' => $message['messageType'] ?? 'text',
                    'status' => $fromMe ? 'sent' : 'received',
                    'timestamp' => \Carbon\Carbon::createFromTimestamp($message['messageTimestamp'])
                ]
            );

            // Update donatur last message
            $donatur->update([
                'last_message_at' => \Carbon\Carbon::createFromTimestamp($message['messageTimestamp'])
            ]);

            // Trigger notification or realtime update
            // broadcast(new NewMessageReceived($donatur, $message));
        }
    }

    private function handleMessageUpdate($data)
    {
        foreach ($data as $update) {
            $messageId = $update['key']['id'];
            $status = $this->mapStatus($update['update']['status'] ?? 0);

            Message::where('whatsapp_message_id', $messageId)
                ->update(['status' => $status]);
        }
    }

    private function handleConnectionUpdate($data)
    {
        // Log connection status changes
        \Log::info('WhatsApp connection update', $data);
    }

    private function mapStatus($statusCode)
    {
        return match($statusCode) {
            0 => 'pending',
            1 => 'sent',
            2 => 'delivered',
            3 => 'read',
            4 => 'played',
            default => 'unknown'
        };
    }
}
